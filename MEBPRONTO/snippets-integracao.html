3️⃣ Sistema de Mensagens - Enviar

Função para enviar mensagem no chat:

async function enviarMensagem(pacienteId, nutricionistaId, texto) {
  try {
    const remetenteId = currentUserId; // Seu ID

    // Enviar para Supabase
    await meb.enviarMensagem(
      pacienteId,
      nutricionistaId,
      remetenteId,
      texto,
      'texto' // tipo: 'texto', 'foto', 'arquivo', 'audio'
    );

    // Limpar input
    document.getElementById('message-input').value = '';

    // Recarregar histórico
    await carregarHistoricoMensagens(pacienteId, nutricionistaId);

    showToast('✓ Mensagem enviada');
  } catch (error) {
    console.error('❌ Erro ao enviar:', error);
    showToast('Erro ao enviar mensagem', 'error');
  }
}

4️⃣ Sistema de Mensagens - Carregar Histórico

Função para carregar chat entre paciente e nutricionista:

async function carregarHistoricoMensagens(pacienteId, nutricionistaId) {
  try {
    const mensagens = await meb.obterMensagens(
      pacienteId,
      nutricionistaId,
      50 // Últimas 50 mensagens
    );

    // Renderizar no DOM
    const chatContainer = document.getElementById('chat-messages');
    chatContainer.innerHTML = mensagens
      .reverse() // Ordem cronológica (mais antiga primeiro)
      .map(m => {
        const isOwn = m.remetente_id === currentUserId;
        const hora = new Date(m.criado_em).toLocaleTimeString('pt-BR', {
          hour: '2-digit',
          minute: '2-digit'
        });

        return `
          

            

              ${m.tipo === 'foto'
                ? ``
                : m.texto}
            

            ${hora}
          

        `;
      })
      .join('');

    // Scroll para o fim
    chatContainer.scrollTop = chatContainer.scrollHeight;

    // Marcar como lidas
    await meb.marcarMensagensLidas(pacienteId, nutricionistaId);
  } catch (error) {
    console.error('❌ Erro ao carregar mensagens:', error);
  }
}

7️⃣ Registrar Peso

Função para registrar peso e calcular progresso:

async function registrarPesoHoje(pesoKg) {
  const pacienteId = currentUserId;

  try {
    // Registrar no Supabase (retorna stats calculadas)
    const result = await meb.registrarPeso(pacienteId, parseFloat(pesoKg));

    // result[0] = {
    //   peso_kg: 74.2,
    //   percentual_progresso: 45.5,
    //   restante_para_meta: 9.2,
    //   dias_no_plano: 67
    // }

    const stats = result[0];

    // Atualizar UI
    document.getElementById('current-weight').textContent = stats.peso_kg.toFixed(1) + ' kg';
    document.getElementById('weight-goal').textContent = stats.restante_para_meta.toFixed(1) + ' kg até a meta';

    // Atualizar barra de progresso
    const progressBar = document.getElementById('progress-bar');
    progressBar.style.width = stats.percentual_progresso + '%';
    progressBar.textContent = Math.round(stats.percentual_progresso) + '%';

    // Animar
    showToast(`✓ ${stats.percentual_progresso}% de progresso!`);
  } catch (error) {
    console.error('❌ Erro:', error);
  }
}

8️⃣ Carregar Cardápio da Semana

Mostrar cardápio personalizado:

async function carregarCardapioSemana(pacienteId) {
  try {
    const cardapio = await meb.obterCardapioSemana(pacienteId);

    // Agrupar por dia da semana
    const porDia = {};
    cardapio.forEach(item => {
      const dia = item.dia_nome;
      if (!porDia[dia]) porDia[dia] = [];
      porDia[dia].push(item);
    });

    // Renderizar
    const container = document.getElementById('cardapio-semana');
    container.innerHTML = Object.entries(porDia).map(([dia, refeicoes]) => `
      

        
${dia}

        ${refeicoes.map(r => `
          

            ${r.tipo_refeicao.replace(/_/g, ' ')}
            
${r.descricao}

            ${r.calorias} kcal
          

        `).join('')}
      

    `).join('');
  } catch (error) {
    console.error('❌ Erro:', error);
  }
}

9️⃣ Registrar Entrada Diária (Diário)

Registrar humor, energia, sintomas:

async function registrarEntradaDiaria() {
  const pacienteId = currentUserId;

  // Coletar do formulário
  const humor = document.getElementById('mood-select').value; // 'muito_feliz', 'feliz', etc
  const energia = parseFloat(document.getElementById('energy-slider').value); // 1-10
  const saciedade = parseFloat(document.getElementById('fullness-slider').value); // 1-10
  const qualidadeSono = parseFloat(document.getElementById('sleep-slider').value); // 1-10
  const notas = document.getElementById('diary-notes').value;
  const sintomasSelecionados = Array.from(
    document.querySelectorAll('input[name="symptoms"]:checked')
  ).map(x => x.value);

  try {
    await meb.registrarDiario(pacienteId, {
      humor,
      energia,
      saciedade,
      qualidadeSono,
      notas,
      sintomas: sintomasSelecionados
    });

    showToast('✓ Entrada do diário salva!');
    document.getElementById('diary-form').reset();
  } catch (error) {
    console.error('❌ Erro:', error);
  }
}

  }

✨ Função Auxiliar - Toast

Você já tem isso no código? Se não, adicione:

function showToast(message, type = 'success') {
  const toast = document.createElement('div');
  toast.className = `toast ${type}`;
  toast.textContent = message;

  document.body.appendChild(toast);

  setTimeout(() => {
    toast.classList.add('show');
  }, 10);

  setTimeout(() => {
    toast.classList.remove('show');
    setTimeout(() => toast.remove(), 300);
  }, 3000);
}
